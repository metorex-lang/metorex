describe "a cart" do
  it "totals" do
    expect(cart.total).to eq(
      42
    )
  end

  it "changes" do
    expect {
      cart.add(1)
    }.to change { cart.size }.by(2)
  end

  it "chains" do
    values = [1, 2]
      .map { |value| value * 2 }
      .sum
    expect(values).to eq(7) if values
  end

  it "loops" do
    while ready do
      step
    end
    if done
      finish
    end
  end
end
